(() => {
  let out = '';
  
  // ============ PASO 1: TIMELINE PRIMERO (para cruzar nombres) ============
  const bar = document.querySelector('.Opta-Timeline-Bar');
  const jugadoresLocal = new Set();
  const jugadoresVisitante = new Set();
  
  const extraerNombresDeEventos = (lista) => {
    if (!lista) return [];
    const nombres = [];
    lista.querySelectorAll('li.Opta-MatchEvent').forEach(li => {
      li.querySelectorAll('p').forEach(p => {
        const t = p.innerText.trim();
        if (t && !t.toLowerCase().startsWith('asistencia') && t.length < 40) {
          nombres.push(t);
        }
      });
    });
    return nombres;
  };
  
  if (bar) {
    extraerNombresDeEventos(bar.querySelector('.Opta-Events.Opta-Home'))
      .forEach(n => jugadoresLocal.add(n));
    extraerNombresDeEventos(bar.querySelector('.Opta-Events.Opta-Away'))
      .forEach(n => jugadoresVisitante.add(n));
  }
  
  const hayTimeline = jugadoresLocal.size > 0 || jugadoresVisitante.size > 0;
  
  // ============ PASO 2: IDENTIFICAR TABLAS ============
  const tablas = [...document.querySelectorAll('table')].filter(t => {
    const h = t.innerText.slice(0, 300);
    return /G\s+A\s+TR\s+TA\s+Crn/.test(h) || (h.includes('SOnT') && h.includes('SAV'));
  });
  
  const tablasEquipos = tablas.filter(t => {
    let el = t.parentElement;
    for (let j = 0; j < 6 && el; j++) {
      if (el.className && typeof el.className === 'string' && el.className.includes('Opta-Team-Both')) {
        return false;
      }
      el = el.parentElement;
    }
    return true;
  });
  
  const tablasConEquipo = tablasEquipos.map(t => {
    const filas = [...t.querySelectorAll('tr')].slice(1);
    const nombres = filas.map(r => r.querySelector('td')?.innerText.trim()).filter(Boolean);
    
    let coincideLocal = 0;
    let coincideVisitante = 0;
    nombres.forEach(n => {
      if (jugadoresLocal.has(n)) coincideLocal++;
      if (jugadoresVisitante.has(n)) coincideVisitante++;
    });
    
    let equipo;
    if (!hayTimeline) equipo = 'FALLBACK';
    else if (coincideLocal > coincideVisitante) equipo = 'LOCAL';
    else if (coincideVisitante > coincideLocal) equipo = 'VISITANTE';
    else equipo = 'FALLBACK';
    
    return { tabla: t, equipo };
  });
  
  const orden = { 'LOCAL': 0, 'VISITANTE': 1, 'FALLBACK': 2 };
  tablasConEquipo.sort((a, b) => orden[a.equipo] - orden[b.equipo]);
  
  // ============ PASO 3: SALIDA DE ESTADÍSTICAS ============
  out += '===== ESTADÍSTICAS DE JUGADORES =====\n\n';
  
  let numEquipo = 1;
  tablasConEquipo.forEach(({tabla}) => {
    out += `--- EQUIPO ${numEquipo++} ---\n`;
    tabla.querySelectorAll('tr').forEach(r => {
      out += Array.from(r.querySelectorAll('th,td'))
        .map(c => c.innerText.trim().replace(/\s+/g, ' '))
        .join('\t') + '\n';
    });
    out += '\n';
  });
  
  // ============ PASO 4: TIMELINE ============
  if (bar) {
    const extraerEventos = (lista) => {
      if (!lista) return [];
      return [...lista.querySelectorAll('li.Opta-MatchEvent')].map(li => {
        const titulo = li.querySelector('.Opta-Event-Title')?.innerText.trim() || '?';
        const minuto = li.querySelector('.Opta-Event-Min')?.innerText.replace(/[^\d+]/g, '').trim() || '?';
        const tipoClase = [...li.classList].find(c => c.startsWith('Opta-Event-Type-')) || '';
        const esSustitucion = tipoClase === 'Opta-Event-Type-990';
        
        const parrafos = [...li.querySelectorAll('p')]
          .map(p => p.innerText.trim())
          .filter(t => t.length > 0);
        
        const asistencia = parrafos
          .filter(p => p.toLowerCase().startsWith('asistencia'))
          .map(p => p.replace(/asistencia\s*:?\s*/i, '').trim())
          .join(', ');
        
        const jugadores = parrafos.filter(p => !p.toLowerCase().startsWith('asistencia'));
        
        let linea;
        if (esSustitucion && jugadores.length >= 2) {
          linea = `${minuto}' | ${titulo} | Sale: ${jugadores[0]} → Entra: ${jugadores[1]}`;
        } else {
          linea = `${minuto}' | ${titulo} | ${jugadores.join(' / ')}`;
        }
        if (asistencia) linea += ` (asist: ${asistencia})`;
        return linea;
      });
    };
    
    const home = extraerEventos(bar.querySelector('.Opta-Events.Opta-Home'));
    const away = extraerEventos(bar.querySelector('.Opta-Events.Opta-Away'));
    
    out += '===== TIMELINE =====\n\n--- LOCAL ---\n' + home.join('\n');
    out += '\n\n--- VISITANTE ---\n' + away.join('\n');
  } else {
    out += '⚠️ No se encontró la timeline\n';
  }
  
  // ============ PASO 5: COPIAR ============
  console.log(out);
  
  const copiar = async () => {
    try {
      await navigator.clipboard.writeText(out);
      console.log('\n✅ Copiado');
    } catch (e) {
      const ta = document.createElement('textarea');
      ta.value = out;
      ta.style.cssText = 'position:fixed;left:-9999px';
      document.body.appendChild(ta);
      ta.focus(); ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
      console.log('\n✅ Copiado (alt)');
    }
  };
  copiar();
})();
